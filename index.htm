<script src="https://cdn.jsdelivr.net/npm/three@0.160.0/build/three.min.js"></script>

<script src="https://cdn.jsdelivr.net/npm/three@0.160.0/examples/js/controls/OrbitControls.js"></script>


<script>

const container = document.getElementById('container');

const WIDTH = 480;

const HEIGHT = 480;


// 场景、相机、渲染器

const scene = new THREE.Scene();

scene.background = new THREE.Color(0x87ceeb);

const camera = new THREE.PerspectiveCamera(55, WIDTH/HEIGHT, 0.1, 1000);

camera.position.set(0, 3.2, 7);


const renderer = new THREE.WebGLRenderer({antialias:true});

renderer.setSize(WIDTH, HEIGHT);

renderer.shadowMap.enabled = true;

renderer.shadowMap.type = THREE.PCFSoftShadowMap;

container.appendChild(renderer.domElement);


// 控制器：拖拽旋转

const controls = new THREE.OrbitControls(camera, renderer.domElement);

controls.enableDamping = true;

controls.dampingFactor = 0.05;

controls.maxPolarAngle = Math.PI * 0.82;

controls.minDistance =3;

controls.maxDistance=12;


//灯光

const dirLight = new THREE.DirectionalLight(0xffffff,1.1);

dirLight.position.set(6,10,4);

dirLight.castShadow=true;

scene.add(dirLight);

scene.add(new THREE.AmbientLight(0xffffff,0.45));


//地面平台 降低高度

const groundGeo = new THREE.PlaneGeometry(22,22);

const groundMat = new THREE.MeshLambertMaterial({color:0x639b47});

const ground = new THREE.Mesh(groundGeo, groundMat);

ground.rotation.x = -Math.PI/2;

ground.receiveShadow=true;

ground.position.y = 0;

scene.add(ground);


//地面网格线

const grid = new THREE.GridHelper(22,22,0x3b6b2b,0x4e8838);

grid.position.y=0.01;

scene.add(grid);


//环境：树木、石块、云朵

function makeTree(x,z){

    const trunkMat = new THREE.MeshLambertMaterial({color:0x8b5a2b});

    const leafMat = new THREE.MeshLambertMaterial({color:0x2d882d});

    const trunk = new THREE.Mesh(new THREE.BoxGeometry(0.6,2.2,0.6),trunkMat);

    trunk.position.set(x,1.1,z);

    trunk.castShadow=true;

    const leaf = new THREE.Mesh(new THREE.BoxGeometry(2.2,2,2.2),leafMat);

    leaf.position.set(x,3.2,z);

    leaf.castShadow=true;

    scene.add(trunk,leaf);

}

makeTree(-7,-6);

makeTree(6,-5);

makeTree(-6,6);

makeTree(7,5);


function makeRock(x,z){

    const mat = new THREE.MeshLambertMaterial({color:0x888888});

    const m = new THREE.Mesh(new THREE.BoxGeometry(1.1,0.8,1.1),mat);

    m.position.set(x,0.4,z);

    m.castShadow=true;

    scene.add(m);

}

makeRock(-4,3);

makeRock(3,-2);


//远处3D云层

function makeCloud(x,y,z){

    const mat = new THREE.MeshLambertMaterial({color:0xffffff,transparent:true,opacity:0.88});

    const g = new THREE.Group();

    for(let i=0;i<6;i++){

        const b = new THREE.Mesh(new THREE.BoxGeometry(1.8,1.2,1.8),mat);

        b.position.set((Math.random()-0.5)*3, (Math.random()-0.5)*1.2, (Math.random()-0.5)*3);

        g.add(b);

    }

    g.position.set(x,y,z);

    scene.add(g);

    return g;

}

const clouds = [

    makeCloud(-12,7,-10),

    makeCloud(14,8,-12),

    makeCloud(-10,9,12),

    makeCloud(13,7,11)

];


// ========= 方块小人 修复穿模，严格拼接 =========

const playerGroup = new THREE.Group();

playerGroup.position.y = 0;

scene.add(playerGroup);


const matSkin = new THREE.MeshLambertMaterial({color:0xf2c397});

const matBlue = new THREE.MeshLambertMaterial({color:0x3366bb});

const matPants = new THREE.MeshLambertMaterial({color:0x224488});


//头

const head = new THREE.Mesh(new THREE.BoxGeometry(1,1,1), matSkin);

head.position.y = 2.6;

head.castShadow=true;

playerGroup.add(head);


//身体躯干

const body = new THREE.Mesh(new THREE.BoxGeometry(1,1.2,0.6), matBlue);

body.position.y =1.5;

body.castShadow=true;

playerGroup.add(body);


//手臂

const armL = new THREE.Mesh(new THREE.BoxGeometry(0.4,1.1,0.4),matSkin);

armL.position.set(-0.72,1.6,0);

armL.castShadow=true;

playerGroup.add(armL);


const armR = new THREE.Mesh(new THREE.BoxGeometry(0.4,1.1,0.4),matSkin);

armR.position.set(0.72,1.6,0);

armR.castShadow=true;

playerGroup.add(armR);


//腿部，无缝贴合

const legL = new THREE.Mesh(new THREE.BoxGeometry(0.4,1.1,0.4),matPants);

legL.position.set(-0.32,0.45,0);

legL.castShadow=true;

playerGroup.add(legL);


const legR = new THREE.Mesh(new THREE.BoxGeometry(0.4,1.1,0.4),matPants);

legR.position.set(0.32,0.45,0);

legR.castShadow=true;

playerGroup.add(legR);


//小人物理状态

const state = {

    isDragging:false,

    floatHeight:0,

    velocityY:0,

    gravity:-18,

    groundY:0,

    squash:1.0, //挤压形变

    staggerTimer:0

};


//拖拽悬浮逻辑

renderer.domElement.addEventListener('pointerdown',()=>{

    state.isDragging=true;

});

renderer.domElement.addEventListener('pointerup',()=>{

    state.isDragging=false;

    //松手开启下落

    state.velocityY = state.floatHeight * 7.5;

});

renderer.domElement.addEventListener('pointerleave',()=>{

    state.isDragging=false;

    state.velocityY = state.floatHeight *7.5;

});


const clock = new THREE.Clock();


function animate(){

    requestAnimationFrame(animate);

    const dt = clock.getDelta();

    controls.update();


    //拖拽悬浮

    if(state.isDragging){

        state.floatHeight += (1.8 - state.floatHeight)* 6 * dt;

        state.velocityY=0;

    }else{

        //重力下落

        state.velocityY += state.gravity * dt;

        state.floatHeight += state.velocityY * dt;


        //地面碰撞

        if(state.floatHeight <=0){

            state.floatHeight =0;

            if(state.velocityY < -3){

                //落地冲击，挤压形变+吃力踉跄动画

                state.squash = 0.62;

                state.staggerTimer = 0.45;

                state.velocityY = -state.velocityY *0.32;

            }else{

                state.velocityY =0;

            }

        }

    }


    //落地恢复形变

    if(state.staggerTimer>0){

        state.staggerTimer -= dt;

    }

    state.squash += (1 - state.squash) * 7 * dt;


    //应用位置+挤压缩放（Y轴挤压模拟吃力）

    playerGroup.position.y = state.floatHeight;

    playerGroup.scale.y = state.squash;


    //云朵缓慢飘动

    clouds.forEach((c,i)=>{

        c.position.x += dt *0.22 * ((i%2)?1:-1);

        if(c.position.x>18) c.position.x=-18;

        if(c.position.x<-18) c.position.x=18;

    });


    renderer.render(scene,camera);

}

animate();

</script>
